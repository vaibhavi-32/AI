#Q1


def heuristic(current, goal):
    x, y = current
    gx, gy = goal
    return abs(gx - x) + abs(gy - y)

# Input
current = (2, 1)
goal = (5, 4)

# Output
print("Current Position:", current)
print("Treasure Position:", goal)
print("Heuristic Distance =", heuristic(current, goal))





#Q2

goal = [
    [1,2,3],
    [4,5,6],
    [7,8,0]
]

current = [
    [1,2,3],
    [4,0,6],
    [7,5,8]
]

def manhattan(current, goal):
    distance = 0

    for num in range(1,9):
        for i in range(3):
            for j in range(3):
                if current[i][j] == num:
                    x1, y1 = i, j

        for i in range(3):
            for j in range(3):
                if goal[i][j] == num:
                    x2, y2 = i, j

        distance += abs(x1-x2) + abs(y1-y2)

    return distance

print("Current Puzzle:")
for row in current:
    print(row)

print("\nGoal Puzzle:")
for row in goal:
    print(row)

print("\nTotal Manhattan Distance =", manhattan(current, goal))




#Q3

room = [
    ['D','C','D'],
    ['C','D','C'],
    ['C','D','D']
]

dirty = 0

for row in room:
    dirty += row.count('D')

print("Room:")
for row in room:
    print(row)

print("\nDirty Cells =", dirty)
print("Heuristic =", dirty)
